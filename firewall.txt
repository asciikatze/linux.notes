-+-+-+-+-+-+ iptables -+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+
# netfilter: kernel framework. iptables: packet selection system.
# default tables: filter, nat, mangle, raw, security
# defailt chains: input, output, forward, prerouting, postrouting
# default targets: accept, drop, reject, log, return
# creating rules = match + target
# ip6tables handles ipv6!

apt install iptables-presistent  #package for presistent rule change!

iptables -L  #list policies
iptables -P {chain} {policy}  #change policy
iptables -P INPUT DROP  #drop all incoming packets
iptables -I INPUT -p tcp --dport 22 -j DROP  #drop all ssh connections

# use iptables -A to append more rules, iptables -D to delete a rule
iptables -A INPUT -p tcp --dport 22 -s {sourceip} -j ACCEPT  #allow ssh from specific ip

# configure webserver http and https
iptables -A INPUT -p tcp --dport 80 -j ACCEPT
iptables -A INPUT -p tcp --dport 443 -j ACCEPT
# allow dns traffic
iptables -A INPUT -p tcp --dport 53 -j ACCEPT
iptables -A INPUT -p udp --dport 53 -j ACCEPT

# configure ftp server
iptables -A INPUT -p tcp --dport 21 -j ACCEPT
iptables -A INPUT -p tcp --dport 20 -j ACCEPT
iptables -A INPUT -m conntrack --cstate ESTABLISHED,RELATED -j ACCEPT

iptables -F  #flush all rules except policies
