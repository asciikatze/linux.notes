-+-+-+-+-+-+ dosfloppy +-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+
# create a blank floppy image
mkfs.msdos -C /path/imagefile.img 1440

# mount
sudo mkdir /media/floppy1/
sudo mount -o loop /path/imagefile.img /media/floppy1/
sudo mount /path/imagefile.img -o umask=0 /media/floppy1/ #use this!
# unmount
sudo umount /media/floppy1/

# send tmp and logfiles to ram
file: /etc/fstab  #add following lines
  tmpfs  /var/log  tmpfs  defaults,noatime 0 0
  tmpfs  /var/tmp  tmpfs  defaults,noatime 0 0
  tmpfs  /tmp      tmpfs  defaults,noatime 0 0

# disable journaling

debugfs -R features /dev/sdb2
tune2fs -O ^has_journal /dev/sdb2  #disable journal; whitour the ^ enable journal
debugfs -R features /dev/sdb2  #see if journaling dissapeared
